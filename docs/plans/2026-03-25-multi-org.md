> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Allow managing multiple GitHub orgs in a single database, with `borg org add/remove/list` subcommands and `--org` filter on all commands.

**Architecture:** Replace single-org `sync_meta` keys with an `orgs` table. Add `org` column to `commits` and `repo_sync` tables. All report/export/status queries gain an optional `WHERE org = ?` filter. `fetch` loops over all registered orgs by default.

```bash
db_org_add() {
    local db="$1" name="$2" since="$3"
    local e_name
    e_name=$(sql_escape "$name")
    local e_since
    e_since=$(sql_escape "$since")
    local existing
    existing=$(sqlite3 "$db" "SELECT COUNT(*) FROM orgs WHERE name = '$e_name';")
    if [[ "$existing" -gt 0 ]]; then
        echo "Org '$name' already registered." >&2
        return 1
    fi
    sqlite3 "$db" "INSERT INTO orgs (name, since_date, added_at) VALUES ('$e_name', '$e_since', '$(utc_now)');"
    echo "Added org '$name' (since $since)."
}

db_org_remove() {
    local db="$1" name="$2"
    local e_name
    e_name=$(sql_escape "$name")
    local existing
    existing=$(sqlite3 "$db" "SELECT COUNT(*) FROM orgs WHERE name = '$e_name';")
    if [[ "$existing" -eq 0 ]]; then
        echo "Org '$name' not found." >&2
        return 1
    fi
    sqlite3 "$db" "DELETE FROM commits WHERE org = '$e_name';
                   DELETE FROM repo_sync WHERE org = '$e_name';
                   DELETE FROM orgs WHERE name = '$e_name';"
    echo "Removed org '$name' and all its data."
}

db_org_list() {
    local db="$1"
    local count
    count=$(sqlite3 "$db" "SELECT COUNT(*) FROM orgs;")
    if [[ "$count" -eq 0 ]]; then
        echo "No orgs registered. Run 'borg org add <name> --since <date>' to add one."
        return
    fi
    printf "  %-25s %-12s %-20s %8s\n" "Org" "Since" "Added" "Commits"
    printf "  ─────────────────────────────────────────────────────────────────\n"
    sqlite3 -separator '|' "$db" \
        "SELECT o.name, o.since_date, o.added_at,
                COALESCE((SELECT COUNT(*) FROM commits c WHERE c.org = o.name), 0)
         FROM orgs o ORDER BY o.name;" | \
    while IFS='|' read -r name since added commits; do
        printf "  %-25s %-12s %-20s %8s\n" "$name" "$since" "${added%T*}" "$commits"
    done
}

db_org_get_all() {
    local db="$1"
    sqlite3 -separator '|' "$db" "SELECT name, since_date FROM orgs ORDER BY name;"
}

db_org_get() {
    local db="$1" name="$2"
    local e_name
    e_name=$(sql_escape "$name")
    sqlite3 -separator '|' "$db" "SELECT name, since_date FROM orgs WHERE name = '$e_name';"
}
```

Actually, the WHERE clause composition is tricky with optional filters. Simpler approach — always use WHERE 1=1 so we can append ANDs:

```bash
db_show_status() {
    local db="$1" org_filter="${2:-}"

    local last_run
    last_run=$(db_get_meta "$db" "last_run_at")

    echo "Borg sync status"
    echo "  Last run:   ${last_run:-never}"

        local e_org
        e_org=$(sql_escape "$org_filter")

    local total enriched detected
    if ! total=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE $org_where;" 2>&1); then
        echo "Error: could not query database: $total" >&2
        return 1
    fi
    enriched=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE $org_where AND additions IS NOT NULL;")
    detected=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE $org_where AND ai_tool IS NOT NULL;")

    echo "  Commits:     $total"
    echo "  Enriched:    $enriched / $total"
    echo "  AI detected: $detected"
```

```bash
fetch_commits() {
    local db="$1" org_filter="${2:-}"

    local org_list
    if [[ -n "$org_filter" ]]; then
        org_list=$(db_org_get "$db" "$org_filter")
        if [[ -z "$org_list" ]]; then
            echo "Error: org '$org_filter' not registered. Run 'borg org add $org_filter --since <date>' first." >&2
            exit 1
        fi
    else
        org_list=$(db_org_get_all "$db")
        if [[ -z "$org_list" ]]; then
            echo "Error: no orgs registered. Run 'borg org add <name> --since <date>' first." >&2
            exit 1
        fi
    fi

    while IFS='|' read -r org since; do
        [[ -z "$org" ]] && continue
        echo ""
        echo "=== Fetching org: $org (since $since) ==="

        echo "Phase 1: Listing commits..."
        local repos
        repos=$(fetch_repo_list "$org") || {
            echo "Error: failed to list repos for org '$org'. Skipping." >&2
            continue
        }
        local repo_count
        repo_count=$(echo "$repos" | grep -c . || true)
        echo "Found $repo_count repos"

        local i=0
        while IFS= read -r repo; do
            [[ -z "$repo" ]] && continue
            i=$((i + 1))
            local count
            count=$(fetch_repo_commits "$db" "$org" "$repo" "$since")
            if [[ "$count" -gt 0 ]]; then
                printf "\r  [%d/%d] %-40s %d new commits\n" "$i" "$repo_count" "$repo" "$count" >&2
            else
                printf "\r  [%d/%d] %-40s                    \r" "$i" "$repo_count" "$repo" >&2
            fi
        done <<< "$repos"
        echo "" >&2
    done <<< "$org_list"

    local total unenriched
    total=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits;")
    unenriched=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE additions IS NULL;")
    echo "Phase 1 complete. $total total commits, $unenriched need enrichment."

    # Phase 2 — enrich all unenriched commits (across all orgs)
    enrich_commits "$db"

    # Update last_run
    db_set_meta "$db" "last_run_at" "$(utc_now)"

    echo "Fetch complete."
}
```

```bash
report_header() {
    local db="$1" title_org="$2" org_where="$3"

    local total_commits ai_commits total_loc ai_loc

    total_commits=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE $org_where;")
    ai_commits=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE $org_where AND ai_confidence = 'high';")
    total_loc=$(sqlite3 "$db" "SELECT COALESCE(SUM($LOC_EXPR),0) FROM commits WHERE $org_where;")
    ai_loc=$(sqlite3 "$db" "SELECT COALESCE(SUM($LOC_EXPR),0) FROM commits WHERE $org_where AND ai_confidence = 'high';")
```

### Task 7: Rewrite borg CLI for multi-org

**Files:**
- Modify: `borg` (complete rewrite of command dispatch)

**Step 1: Rewrite the CLI**

The new CLI structure:

```bash
#!/usr/bin/env bash
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
DEFAULT_DB="$SCRIPT_DIR/data/tracker.db"

source "$SCRIPT_DIR/lib/db.sh"
source "$SCRIPT_DIR/lib/fetch.sh"
source "$SCRIPT_DIR/lib/detect.sh"
source "$SCRIPT_DIR/lib/report.sh"

usage() {
    cat <<EOF
borg — AI commit adoption tracker
"Resistance is futile"

Usage:
  borg org add <name> [--since <date>]
  borg org remove <name> [--force]
  borg org list
  borg fetch  [--org <name>] [--db <path>]
  borg report [--org <name>] [--top <n>] [--db <path>]
  borg export [--org <name>] [--csv <path>] [--db <path>]
  borg status [--org <name>] [--db <path>]

Commands:
  org add     Register a GitHub organization
  org remove  Unregister org and delete its data
  org list    Show registered organizations
  fetch       Fetch commits (all orgs, or --org for one)
  report      Show AI adoption report with rankings and trends
  export      Export database to CSV
  status      Show sync status

Options:
  --org    Filter to a specific organization
  --since  Floor date YYYY-MM-DD (for org add; default: 90 days ago)
  --top    Number of entries in rankings (default: 10)
  --db     Database path (default: <script_dir>/data/tracker.db)
  --csv    CSV output path (default: stdout)
  --force  Skip confirmation on destructive operations
EOF
    exit 1
}

if [[ $# -eq 0 ]]; then
    usage
fi

COMMAND="$1"
shift

# Handle "org" subcommand
if [[ "$COMMAND" == "org" ]]; then
    if [[ $# -eq 0 ]]; then
        echo "Error: org requires a subcommand (add, remove, list)" >&2
        usage
    fi
    SUBCOMMAND="$1"
    shift
    COMMAND="org_$SUBCOMMAND"
fi

DB="$DEFAULT_DB"
ORG=""
SINCE=""
TOP=10
CSV=""
FORCE=false
ORG_NAME=""

# For org add/remove, first positional arg is the org name
if [[ "$COMMAND" == "org_add" || "$COMMAND" == "org_remove" ]]; then
    if [[ $# -eq 0 ]]; then
        echo "Error: org $SUBCOMMAND requires an org name" >&2
        exit 1
    fi
    ORG_NAME="$1"
    shift
fi

while [[ $# -gt 0 ]]; do
    case "$1" in
        --org)
            [[ $# -ge 2 ]] || { echo "Error: --org requires a value" >&2; exit 1; }
            ORG="$2"; shift 2 ;;
        --since)
            [[ $# -ge 2 ]] || { echo "Error: --since requires a value" >&2; exit 1; }
            SINCE="$2"; shift 2 ;;
        --top)
            [[ $# -ge 2 ]] || { echo "Error: --top requires a value" >&2; exit 1; }
            TOP="$2"; shift 2 ;;
        --db)
            [[ $# -ge 2 ]] || { echo "Error: --db requires a value" >&2; exit 1; }
            DB="$2"; shift 2 ;;
        --csv)
            [[ $# -ge 2 ]] || { echo "Error: --csv requires a value" >&2; exit 1; }
            CSV="$2"; shift 2 ;;
        --force)
            FORCE=true; shift ;;
        *)
            echo "Unknown option: $1" >&2; usage ;;
    esac
done

# Validate --top
if ! [[ "$TOP" =~ ^[0-9]+$ ]] || [[ "$TOP" -eq 0 ]]; then
    echo "Error: --top must be a positive integer" >&2
    exit 1
fi

# Validate --since if provided
if [[ -n "$SINCE" ]] && ! [[ "$SINCE" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}$ ]]; then
    echo "Error: --since must be in YYYY-MM-DD format" >&2
    exit 1
fi

# Initialize database
db_init "$DB"

case "$COMMAND" in
    org_add)
        if [[ -z "$SINCE" ]]; then
            # Default: 90 days ago
            SINCE=$(date -u -v-90d +%Y-%m-%d 2>/dev/null || date -u -d '90 days ago' +%Y-%m-%d)
        fi
        db_org_add "$DB" "$ORG_NAME" "$SINCE"
        ;;
    org_remove)
        if [[ "$FORCE" != true ]]; then
            if command -v gum &>/dev/null; then
                gum confirm "Remove org '$ORG_NAME' and all its data?" || { echo "Cancelled."; exit 0; }
            else
                printf "Remove org '%s' and all its data? [y/N] " "$ORG_NAME"
                read -r confirm
                [[ "$confirm" =~ ^[Yy] ]] || { echo "Cancelled."; exit 0; }
            fi
        fi
        db_org_remove "$DB" "$ORG_NAME"
        ;;
    org_list)
        db_org_list "$DB"
        ;;
    fetch)
        fetch_commits "$DB" "$ORG"
        detect_ai "$DB"
        ;;
    report)
        report_show "$DB" "$ORG" "$TOP"
        ;;
    export)
        if [[ -z "$CSV" ]]; then
            CSV="/dev/stdout"
        fi
        db_export_csv "$DB" "$CSV" "$ORG"
        ;;
    status)
        db_show_status "$DB" "$ORG"
        ;;
    *)
        echo "Unknown command: $COMMAND" >&2
        usage
        ;;
esac
```
