**Step 4: Create the main entry point `borg`**

```bash
#!/usr/bin/env bash
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
DEFAULT_DB="$SCRIPT_DIR/data/tracker.db"

source "$SCRIPT_DIR/lib/db.sh"
source "$SCRIPT_DIR/lib/fetch.sh"
source "$SCRIPT_DIR/lib/detect.sh"
source "$SCRIPT_DIR/lib/report.sh"

usage() {
    cat <<EOF
borg — AI commit adoption tracker
"Resistance is futile"

Usage:
  borg fetch  --org <org> [--since <date>] [--db <path>]
  borg report [--top <n>] [--db <path>]
  borg export [--csv <path>] [--db <path>]
  borg status [--db <path>]

Commands:
  fetch    Fetch commits from GitHub org (incremental, rate-limit aware)
  report   Show AI adoption report with rankings and trends
  export   Export database to CSV
  status   Show sync status

Options:
  --org    GitHub organization (required on first fetch, remembered after)
  --since  Floor date YYYY-MM-DD (required on first fetch, remembered after)
  --top    Number of entries in rankings (default: 10)
  --db     Database path (default: ./data/tracker.db)
  --csv    CSV output path (default: stdout)
EOF
    exit 1
}

# Parse command
COMMAND="${1:-}"
shift || true

if [[ -z "$COMMAND" ]]; then
    usage
fi

# Parse flags
DB="$DEFAULT_DB"
ORG=""
SINCE=""
TOP=10
CSV=""

while [[ $# -gt 0 ]]; do
    case "$1" in
        --org)   ORG="$2"; shift 2 ;;
        --since) SINCE="$2"; shift 2 ;;
        --top)   TOP="$2"; shift 2 ;;
        --db)    DB="$2"; shift 2 ;;
        --csv)   CSV="$2"; shift 2 ;;
        *)       echo "Unknown option: $1"; usage ;;
    esac
done

# Initialize database
db_init "$DB"

# Resolve org and since from sync_meta if not provided
if [[ -z "$ORG" ]]; then
    ORG=$(db_get_meta "$DB" "org")
fi
if [[ -z "$SINCE" ]]; then
    SINCE=$(db_get_meta "$DB" "default_since")
fi

case "$COMMAND" in
    fetch)
        if [[ -z "$ORG" ]]; then
            echo "Error: --org required on first run"
            exit 1
        fi
        if [[ -z "$SINCE" ]]; then
            echo "Error: --since required on first run (e.g. --since 2026-01-01)"
            exit 1
        fi
        db_set_meta "$DB" "org" "$ORG"
        db_set_meta "$DB" "default_since" "$SINCE"
        fetch_commits "$DB" "$ORG" "$SINCE"
        detect_ai "$DB"
        ;;
    report)
        if [[ -z "$ORG" ]]; then
            echo "Error: no data yet. Run 'borg fetch' first."
            exit 1
        fi
        report_show "$DB" "$ORG" "$SINCE" "$TOP"
        ;;
    export)
        if [[ -z "$CSV" ]]; then
            CSV="/dev/stdout"
        fi
        db_export_csv "$DB" "$CSV"
        ;;
    status)
        db_show_status "$DB"
        ;;
    *)
        echo "Unknown command: $COMMAND"
        usage
        ;;
esac
```

### Task 3: Fetch layer — Phase 1: list commits (lib/fetch.sh)

**Step 1: Implement rate limit checking**

```bash
check_rate_limit() {
    local remaining reset_at
    remaining=$(gh api rate_limit --jq '.rate.remaining')
    reset_at=$(gh api rate_limit --jq '.rate.reset')
    echo "$remaining $reset_at"
}

wait_for_rate_limit() {
    local min_remaining="${1:-200}"
    while true; do
        local info
        info=$(check_rate_limit)
        local remaining="${info%% *}"
        local reset_epoch="${info##* }"
        if [[ "$remaining" -ge "$min_remaining" ]]; then
            echo "$remaining"
            return
        fi
        local now
        now=$(date +%s)
        local wait_seconds=$(( reset_epoch - now + 5 ))
        if [[ "$wait_seconds" -lt 0 ]]; then wait_seconds=5; fi
        local reset_human
        reset_human=$(date -r "$reset_epoch" "+%H:%M:%S %Z" 2>/dev/null || date -d "@$reset_epoch" "+%H:%M:%S %Z" 2>/dev/null || echo "soon")
        echo "Rate limit hit. $remaining remaining. Waiting until $reset_human ($wait_seconds sec)..." >&2
        sleep "$wait_seconds"
    done
}
```

**Step 2: Implement repo listing**

```bash
fetch_repo_list() {
    local org="$1"
    gh repo list "$org" --limit 500 --json name,isArchived \
        --jq '.[] | select(.isArchived == false) | .name'
}
```

**Step 3: Implement Phase 1 — list commits per repo**

For each repo:
1. Check `repo_sync.last_commit_date` to get the `since` parameter
2. If no prior sync, use the global `--since` date
3. Paginate through `gh api repos/{org}/{repo}/commits?since={date}&per_page=100`
4. For each commit: `INSERT OR IGNORE INTO commits (sha, repo, author, email, date, message, fetched_at)`
5. Update `repo_sync` with the newest commit date

Process repos sequentially (Phase 1 is cheap — ~1 API call per repo page).

**Step 4: Wire into fetch_commits**

```bash
fetch_commits() {
    local db="$1" org="$2" since="$3"

    echo "Phase 1: Listing commits..."
    local repos
    repos=$(fetch_repo_list "$org")
    local repo_count
    repo_count=$(echo "$repos" | wc -l | tr -d ' ')
    echo "Found $repo_count repos"

    local i=0
    while IFS= read -r repo; do
        i=$((i + 1))
        fetch_repo_commits "$db" "$org" "$repo" "$since"
        printf "\r  Listing: %d/%d repos  " "$i" "$repo_count" >&2
    done <<< "$repos"
    echo "" >&2

    local total
    total=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits;")
    local unenriched
    unenriched=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE additions IS NULL;")
    echo "Phase 1 complete. $total total commits, $unenriched need enrichment."

    echo "Phase 2: Enriching with stats..."
    enrich_commits "$db" "$org"

    echo "Done."
}
```

### Task 4: Fetch layer — Phase 2: enrich with stats (lib/fetch.sh)

**Files:**
- Modify: `lib/fetch.sh`

**Step 1: Implement enrich_commits**

```bash
enrich_commits() {
    local db="$1" org="$2"

    local total
    total=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE additions IS NULL;")

    if [[ "$total" -eq 0 ]]; then
        echo "All commits already enriched."
        return
    fi

    local done_count=0

    while true; do
        # Get batch of unenriched commits
        local batch
        batch=$(sqlite3 -separator '|' "$db" \
            "SELECT sha, repo FROM commits WHERE additions IS NULL LIMIT 50;")

        if [[ -z "$batch" ]]; then
            break
        fi

        # Check rate limit and determine parallelism
        local remaining
        remaining=$(wait_for_rate_limit 200)
        local parallel=8
        if [[ "$remaining" -lt 500 ]]; then
            parallel=4
        fi

        # Process batch
        while IFS='|' read -r sha repo; do
            enrich_single_commit "$db" "$org" "$repo" "$sha" &

            # Limit parallelism
            local running
            running=$(jobs -r | wc -l)
            while [[ "$running" -ge "$parallel" ]]; do
                wait -n 2>/dev/null || true
                running=$(jobs -r | wc -l)
            done

            done_count=$((done_count + 1))
            printf "\r  Enriching: %d/%d commits  " "$done_count" "$total" >&2
        done <<< "$batch"

        # Wait for remaining background jobs
        wait
    done

    echo "" >&2
    echo "Enrichment complete. $done_count commits updated."
}
```

**Step 2: Implement enrich_single_commit**

```bash
enrich_single_commit() {
    local db="$1" org="$2" repo="$3" sha="$4"

    local result
    result=$(gh api "repos/$org/$repo/commits/$sha" \
        --jq '{additions: .stats.additions, deletions: .stats.deletions}' 2>/dev/null) || return 0

    local additions deletions
    additions=$(echo "$result" | jq -r '.additions // 0')
    deletions=$(echo "$result" | jq -r '.deletions // 0')

    sqlite3 "$db" "UPDATE commits SET additions = $additions, deletions = $deletions WHERE sha = '$sha';"
}
```

Note: parallel writes to SQLite need WAL mode (set in db_init). Serialization is handled by SQLite's built-in locking.

**Step 1: Implement detect_ai**

Run all detection rules as SQL UPDATEs. First reset all ai_tool/ai_confidence to NULL, then apply rules in priority order. Use `WHERE ai_tool IS NULL` on each rule so first-match-wins.

```bash
detect_ai() {
    local db="$1"

    echo "Detecting AI-assisted commits..."

    # Reset detection (allows re-running with updated rules)
    sqlite3 "$db" "UPDATE commits SET ai_tool = NULL, ai_confidence = NULL;"

    # Claude: Co-Authored-By trailer or anthropic email
    sqlite3 "$db" "UPDATE commits SET ai_tool = 'claude', ai_confidence = 'high'
        WHERE ai_tool IS NULL AND (
            message LIKE '%Co-Authored-By:%Claude%'
            OR message LIKE '%Co-authored-by:%Claude%'
            OR message LIKE '%noreply@anthropic.com%'
        );"

    # Copilot: trailer or bot author
    sqlite3 "$db" "UPDATE commits SET ai_tool = 'copilot', ai_confidence = 'high'
        WHERE ai_tool IS NULL AND (
            message LIKE '%Co-Authored-By:%Copilot%'
            OR message LIKE '%Co-authored-by:%Copilot%'
            OR author = 'copilot[bot]'
            OR email LIKE '%copilot%'
        );"

    # Cursor
    sqlite3 "$db" "UPDATE commits SET ai_tool = 'cursor', ai_confidence = 'high'
        WHERE ai_tool IS NULL AND (
            message LIKE '%Co-Authored-By:%Cursor%'
            OR message LIKE '%Co-authored-by:%Cursor%'
        );"

    # Aider
    sqlite3 "$db" "UPDATE commits SET ai_tool = 'aider', ai_confidence = 'high'
        WHERE ai_tool IS NULL AND author LIKE '%(aider)%';"

    # ChatGPT / OpenAI
    sqlite3 "$db" "UPDATE commits SET ai_tool = 'chatgpt', ai_confidence = 'high'
        WHERE ai_tool IS NULL AND (
            message LIKE '%Co-Authored-By:%ChatGPT%'
            OR message LIKE '%Co-authored-by:%ChatGPT%'
            OR message LIKE '%Co-Authored-By:%OpenAI%'
            OR message LIKE '%Co-authored-by:%OpenAI%'
        );"

    # Devin
    sqlite3 "$db" "UPDATE commits SET ai_tool = 'devin', ai_confidence = 'high'
        WHERE ai_tool IS NULL AND (
            message LIKE '%Co-Authored-By:%Devin%'
            OR message LIKE '%Co-authored-by:%Devin%'
            OR author = 'devin-ai[bot]'
        );"

    # Cody
    sqlite3 "$db" "UPDATE commits SET ai_tool = 'cody', ai_confidence = 'high'
        WHERE ai_tool IS NULL AND (
            message LIKE '%Co-Authored-By:%Cody%'
            OR message LIKE '%Co-authored-by:%Cody%'
        );"

    # Amazon Q
    sqlite3 "$db" "UPDATE commits SET ai_tool = 'amazon-q', ai_confidence = 'high'
        WHERE ai_tool IS NULL AND (
            message LIKE '%Co-Authored-By:%Amazon Q%'
            OR message LIKE '%Co-authored-by:%Amazon Q%'
        );"

    # Windsurf
    sqlite3 "$db" "UPDATE commits SET ai_tool = 'windsurf', ai_confidence = 'high'
        WHERE ai_tool IS NULL AND (
            message LIKE '%Co-Authored-By:%Windsurf%'
            OR message LIKE '%Co-authored-by:%Windsurf%'
        );"

    # Codeium
    sqlite3 "$db" "UPDATE commits SET ai_tool = 'codeium', ai_confidence = 'high'
        WHERE ai_tool IS NULL AND (
            message LIKE '%Co-Authored-By:%Codeium%'
            OR message LIKE '%Co-authored-by:%Codeium%'
        );"

    # Tabnine
    sqlite3 "$db" "UPDATE commits SET ai_tool = 'tabnine', ai_confidence = 'high'
        WHERE ai_tool IS NULL AND (
            message LIKE '%Co-Authored-By:%Tabnine%'
            OR message LIKE '%Co-authored-by:%Tabnine%'
        );"

    # Bulk addition heuristic (low confidence)
    sqlite3 "$db" "UPDATE commits SET ai_tool = 'unknown', ai_confidence = 'low'
        WHERE ai_tool IS NULL
            AND additions IS NOT NULL
            AND additions > 100
            AND deletions < 10;"

    # Report counts
    local total ai_high ai_low
    total=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits;")
    ai_high=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE ai_confidence = 'high';")
    ai_low=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE ai_confidence = 'low';")

    echo "Detection complete: $ai_high high-confidence, $ai_low low-confidence, out of $total total."
}
```

```bash
report_header() {
    local db="$1" org="$2" since="$3"

    local total ai_commits total_loc ai_loc
    total=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits;")
    ai_commits=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE ai_confidence = 'high';")
    total_loc=$(sqlite3 "$db" "SELECT COALESCE(SUM(COALESCE(additions,0) + COALESCE(deletions,0)), 0) FROM commits;")
    ai_loc=$(sqlite3 "$db" "SELECT COALESCE(SUM(COALESCE(additions,0) + COALESCE(deletions,0)), 0) FROM commits WHERE ai_confidence = 'high';")

    local ai_pct=0 loc_pct=0
    if [[ "$total" -gt 0 ]]; then
        ai_pct=$(( ai_commits * 100 / total ))
    fi
    if [[ "$total_loc" -gt 0 ]]; then
        loc_pct=$(( ai_loc * 100 / total_loc ))
    fi

    gum style --border double --padding "0 2" --border-foreground 212 \
        "Borg — AI Commit Adoption — $org — since $since" \
        "" \
        "Total commits: $(printf '%,d' $total)    AI-assisted: $(printf '%,d' $ai_commits) (${ai_pct}%)" \
        "Total LOC changed: $(printf '%,d' $total_loc)    AI-assisted LOC: $(printf '%,d' $ai_loc) (${loc_pct}%)"
}
```

```sql
SELECT author,
       COUNT(*) FILTER (WHERE ai_confidence = 'high') as ai_commits,
```

```bash
report_trend() {
    local db="$1"

    echo ""
    echo "Monthly trend:"

    local data
    data=$(sqlite3 -separator '|' "$db" \
        "SELECT strftime('%Y-%m', date) as month,
                COUNT(*) as total,
                COUNT(*) FILTER (WHERE ai_confidence = 'high') as ai,
                COALESCE(SUM(COALESCE(additions,0) + COALESCE(deletions,0)), 0) as total_loc,
                COALESCE(SUM(CASE WHEN ai_confidence = 'high' THEN COALESCE(additions,0) + COALESCE(deletions,0) END), 0) as ai_loc
         FROM commits
         GROUP BY month
         ORDER BY month;")

    # Collect percentages for sparkline
    local pcts=()

    while IFS='|' read -r month total ai total_loc ai_loc; do
        local pct=0 loc_pct=0
        if [[ "$total" -gt 0 ]]; then pct=$(( ai * 100 / total )); fi
        if [[ "$total_loc" -gt 0 ]]; then loc_pct=$(( ai_loc * 100 / total_loc )); fi
        pcts+=("$pct")
        printf "  %s   %5d/%5d  %3d%%  |  %'9d/%'9d LOC  %3d%%\n" \
            "$month" "$ai" "$total" "$pct" "$ai_loc" "$total_loc" "$loc_pct"
    done <<< "$data"

    # Sparkline
    if command -v "$SCRIPT_DIR/deps/spark" &>/dev/null; then
        local sparkline
```

```bash
db_show_status() {
    local db="$1"

    local org since last_run
    org=$(db_get_meta "$db" "org")
    since=$(db_get_meta "$db" "default_since")
    last_run=$(db_get_meta "$db" "last_run_at")

    echo "Borg sync status"
    echo "  Org:        ${org:-not set}"
    echo "  Since:      ${since:-not set}"
    echo "  Last run:   ${last_run:-never}"
    echo ""

    local total enriched detected
    total=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits;" 2>/dev/null || echo 0)
    enriched=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE additions IS NOT NULL;" 2>/dev/null || echo 0)
    detected=$(sqlite3 "$db" "SELECT COUNT(*) FROM commits WHERE ai_tool IS NOT NULL;" 2>/dev/null || echo 0)

    echo "  Commits:    $total"
    echo "  Enriched:   $enriched / $total"
    echo "  AI detected: $detected"
    echo ""

    echo "  Repos synced:"
    sqlite3 -separator '|' "$db" \
        "SELECT repo, commit_count, last_commit_date FROM repo_sync ORDER BY commit_count DESC LIMIT 20;" 2>/dev/null | \
    while IFS='|' read -r repo count last_date; do
        printf "    %-35s %5d commits  (last: %s)\n" "$repo" "$count" "$last_date"
    done
}
```
